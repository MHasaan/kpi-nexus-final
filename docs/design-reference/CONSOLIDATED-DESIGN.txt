   spec §12.3; only values are retunable). This file mirrors those tokens as
   ready-to-use CSS custom properties for design artifacts + prototypes, and
   adds semantic type roles (h1, h2, body, code, …).

   Accent is sky → azure (light blue). Surfaces are cool-neutral (hue 240). Status colors

/* ============================================================================
   Icon — renders Lucide icons from the UMD global (window.lucide), as inline
   SVG via innerHTML so there is no React/DOM reconciliation conflict.
   ============================================================================ */
function Icon({ name, size = 18, strokeWidth = 2, style, className }) {
  const pascal = name.split('-').map((s) => s[0].toUpperCase() + s.slice(1)).join('');
  const lib = window.lucide || {};
  const node = (lib.icons && lib.icons[pascal]) || lib[pascal];
  let inner = '';
  if (node) {
    // Lucide UMD icon node shape: ["svg", {attrs}, [ [tag, attrs], ... ]]
    const children = Array.isArray(node) && Array.isArray(node[2]) ? node[2] : Array.isArray(node) ? node : node.icon || [];
    inner = children
      .map(([tag, attrs]) => {
        const a = Object.entries(attrs || {})
          .map(([k, v]) => `${k}="${v}"`)
          .join(' ');
        return `<${tag} ${a}></${tag}>`;
      })
      .join('');
  }
  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${size}" height="${size}" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="${strokeWidth}" stroke-linecap="round" stroke-linejoin="round">${inner}</svg>`;
  return (
    <span
      className={className}
      style={{ display: 'inline-flex', alignItems: 'center', justifyContent: 'center', flexShrink: 0, ...style }}
      dangerouslySetInnerHTML={{ __html: svg }}
    />
  );
}

function Sparkline({ data, color = 'hsl(var(--accent-primary))', w = 120, h = 36, fill = true, fluid = false }) {
  const min = Math.min(...data), max = Math.max(...data);
  const rng = max - min || 1;
  const pts = data.map((d, i) => [(i / (data.length - 1)) * w, h - 3 - ((d - min) / rng) * (h - 6)]);
  const line = pts.map((p) => p.join(',')).join(' ');
  const area = `${line} ${w},${h} 0,${h}`;
  return (
    <svg viewBox={`0 0 ${w} ${h}`} width={fluid ? '100%' : w} height={h} preserveAspectRatio="none" style={{ display: 'block', width: fluid ? '100%' : undefined }}>
      {fill && <polyline points={area} fill={color} opacity="0.08" stroke="none" />}
      <polyline points={line} fill="none" stroke={color} strokeWidth="2" strokeLinejoin="round" strokeLinecap="round" />
    </svg>
  );
}

function LineChart({ series, w = 640, h = 240, showBand }) {
  const pad = { l: 38, r: 14, t: 14, b: 26 };
  const iw = w - pad.l - pad.r, ih = h - pad.t - pad.b;
  const all = series.flatMap((s) => s.data).filter((v) => v != null && !Number.isNaN(v));
  const min = Math.min(...all, showBand ? Math.min(...showBand.lo) : Infinity);
  const max = Math.max(...all, showBand ? Math.max(...showBand.hi) : -Infinity);
  const rng = max - min || 1;
  const n = series[0].data.length;
  const x = (i) => pad.l + (i / (n - 1)) * iw;
  const y = (v) => pad.t + ih - ((v - min) / rng) * ih;
  const yticks = 4;
  return (
    <svg viewBox={`0 0 ${w} ${h}`} width="100%" style={{ display: 'block', fontFamily: 'var(--font-mono)' }}>
      {Array.from({ length: yticks + 1 }, (_, i) => {
        const v = min + (rng * i) / yticks;
        const yy = y(v);
        return (
          <g key={i}>
            <line x1={pad.l} x2={w - pad.r} y1={yy} y2={yy} stroke="var(--c-divider)" strokeWidth="1" />
            <text x={pad.l - 7} y={yy + 3} textAnchor="end" fontSize="9" fill="var(--c-muted)">{Math.round(v)}</text>
          </g>
        );
      })}
      {showBand && (
        <polygon
          points={[...showBand.hi.map((v, i) => `${x(i)},${y(v)}`), ...showBand.lo.map((v, i) => `${x(i)},${y(v)}`).reverse()].join(' ')}
          fill="hsl(var(--accent-primary) / 0.12)" stroke="none"
        />
      )}
      {series.map((s, si) => {
        const pts = s.data.map((v, i) => [i, v]).filter(([, v]) => v != null && !Number.isNaN(v));
        if (!pts.length) return null;
        const line = pts.map(([i, v]) => `${x(i)},${y(v)}`).join(' ');
        const fi = pts[0][0], li = pts[pts.length - 1][0];
        return (
          <g key={si}>
            {s.fill && <polygon points={`${line} ${x(li)},${pad.t + ih} ${x(fi)},${pad.t + ih}`} fill={s.color} opacity="0.07" />}
            <polyline points={line} fill="none" stroke={s.color} strokeWidth={s.dashed ? 2 : 2.4} strokeDasharray={s.dashed ? '5 4' : 'none'} strokeLinejoin="round" strokeLinecap="round" />
            {s.markers && s.data.map((v, i) => s.markers.includes(i) && v != null ? <circle key={i} cx={x(i)} cy={y(v)} r="4" fill="hsl(var(--status-critical))" stroke="var(--c-surface-bg)" strokeWidth="1.5" /> : null)}
          </g>
        );
      })}
    </svg>
  );
}

function BarChart({ data, color = 'hsl(var(--accent-primary))', w = 640, h = 200 }) {
  const pad = { l: 34, r: 12, t: 12, b: 24 };
  const iw = w - pad.l - pad.r, ih = h - pad.t - pad.b;
  const max = Math.max(...data.map((d) => d.v)) || 1;
  const bw = (iw / data.length) * 0.62;
  return (
    <svg viewBox={`0 0 ${w} ${h}`} width="100%" style={{ display: 'block', fontFamily: 'var(--font-mono)' }}>
      {[0, 0.5, 1].map((t, i) => {
        const yy = pad.t + ih - t * ih;
        return <line key={i} x1={pad.l} x2={w - pad.r} y1={yy} y2={yy} stroke="var(--c-divider)" />;
      })}
      {data.map((d, i) => {
        const bh = (d.v / max) * ih;
        const cx = pad.l + (iw / data.length) * (i + 0.5);
        return (
          <g key={i}>
            <rect x={cx - bw / 2} y={pad.t + ih - bh} width={bw} height={bh} rx="3" fill={d.color || color} />
            <text x={cx} y={h - 8} textAnchor="middle" fontSize="9" fill="var(--c-muted)">{d.label}</text>
          </g>
        );
      })}
    </svg>
  );
}

function Tabs({ tabs, value, onChange }) {
  return (
    <div style={{ display: 'flex', gap: 2, borderBottom: '1px solid var(--c-border)', margin: '4px 0 18px', overflowX: 'auto' }}>
      {tabs.map((t) => (
        <button key={t} onClick={() => onChange && onChange(t)}
          style={{ height: 38, padding: '0 14px', border: 'none', background: 'transparent', cursor: 'pointer', fontSize: 13, fontWeight: 600,
            fontFamily: 'var(--font-sans)', color: value === t ? 'var(--c-strong)' : 'var(--c-muted)',

      ))}
    </div>
  );
}

// pill-style sub navigation (route children)
function SubNav({ items, value, onChange }) {
  return (

          style={{ height: 30, padding: '0 12px', borderRadius: 999, cursor: 'pointer', fontSize: 12.5, fontWeight: 600, fontFamily: 'var(--font-sans)',

      ))}
    </div>
  );
}

/* global React, ReactDOM, Sidebar, Header, CommandPalette, DashboardView, KpisView, KpiDetailView, DashboardsView, AskView, StubView */
const { useState: useStateA, useEffect: useEffectA } = React;

const TITLES = {
  dashboard: { title: 'Dashboard', crumbs: ['Home'] },
  kpis: { title: 'KPIs', crumbs: ['Workspace'] },
  kpiDetail: { title: 'KPI detail', crumbs: ['Workspace', 'KPIs'] },
  dashboards: { title: 'Dashboards', crumbs: ['Workspace'] },
  alerts: { title: 'Alerts', crumbs: ['Workspace'] },
  insights: { title: 'Insights', crumbs: ['Intelligence'] },
  ask: { title: 'Ask', crumbs: ['Intelligence'] },
  okrs: { title: 'OKRs', crumbs: ['Execute'] },
  tasks: { title: 'Tasks', crumbs: ['Execute'] },
  approvals: { title: 'Approvals', crumbs: ['Execute'] },
  reports: { title: 'Reports', crumbs: ['Execute'] },
  people: { title: 'People & org', crumbs: ['Organize'] },
  integrations: { title: 'Integrations', crumbs: ['Organize'] },
  activity: { title: 'Activity', crumbs: ['Organize'] },
  settings: { title: 'Settings', crumbs: ['Organize'] },
};

function App() {
  const [theme, setTheme] = useStateA(() => localStorage.getItem('kn-theme') || 'light');
  const [view, setView] = useStateA('dashboard');
  const [collapsed, setCollapsed] = useStateA(false);
  const [paletteOpen, setPaletteOpen] = useStateA(false);
  const [activeKpi, setActiveKpi] = useStateA(null);

  useEffectA(() => {
    document.documentElement.classList.toggle('dark', theme === 'dark');
    localStorage.setItem('kn-theme', theme);
  }, [theme]);

  useEffectA(() => {
    const onKey = (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === 'k') { e.preventDefault(); setPaletteOpen((p) => !p); }
      if (e.key === 'Escape') setPaletteOpen(false);
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, []);

  const openKpi = (k) => { setActiveKpi(k); setView('kpiDetail'); };
  const navTo = (v) => { setView(v); };

  const meta = TITLES[view] || TITLES.dashboard;

  let content;
  switch (view) {
    case 'dashboard': content = <DashboardView openKpi={openKpi} />; break;
    case 'kpis': content = <KpisView openKpi={openKpi} />; break;
    case 'kpiDetail': content = <KpiDetailView kpi={activeKpi} back={() => setView('kpis')} />; break;
    case 'dashboards': content = <DashboardsView />; break;
    case 'ask': content = <AskView />; break;
    case 'alerts': content = <AlertsView />; break;
    case 'insights': content = <InsightsView />; break;
    case 'okrs': content = <OkrsView />; break;
    case 'tasks': content = <TasksView />; break;
    case 'reports': content = <ReportsView />; break;
    case 'approvals': content = <ApprovalsView />; break;
    case 'activity': content = <ActivityView />; break;
    case 'people': content = <PeopleView />; break;
    case 'integrations': content = <IntegrationsView />; break;
    case 'settings': content = <SettingsView />; break;
    default: content = <DashboardView openKpi={openKpi} />;
  }

  return (
    <div style={{ display: 'flex', height: '100vh', overflow: 'hidden', background: 'var(--c-surface-bg)' }}>
      <Sidebar view={view === 'kpiDetail' ? 'kpis' : view} setView={navTo} collapsed={collapsed} />
      <div style={{ flex: 1, display: 'flex', flexDirection: 'column', minWidth: 0 }}>
        <Header
          title={meta.title}
          crumbs={meta.crumbs}
          theme={theme}
          toggleTheme={() => setTheme((t) => (t === 'dark' ? 'light' : 'dark'))}
          onSearch={() => setPaletteOpen(true)}
          onToggleSidebar={() => setCollapsed((c) => !c)}
        />
        <main style={{ flex: 1, overflowY: 'auto', padding: '24px 28px 48px' }}>
          <div key={view} className="kn-view" style={{ maxWidth: view === 'ask' ? 860 : 1180, margin: '0 auto' }}>{content}</div>
        </main>
      </div>
      <CommandPalette open={paletteOpen} onClose={() => setPaletteOpen(false)} setView={navTo} />
    </div>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App />);
